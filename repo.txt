def read_report():

    if not REPORT.exists():
        return None

    try:

        return json.loads(
            REPORT.read_text(encoding="utf-8")
        )

    except (OSError, json.JSONDecodeError):

        return None


def read_source(path):

    if not path.exists():
        return None

    try:

        return path.read_text(encoding="utf-8")

    except OSError:

        return None


def run_pipeline():

    stages = []
    findings = []
    result = None

    try:

        # ----------------------------------------------------
        # STAGE 01: SOURCE ANALYSIS
        # ----------------------------------------------------

        with st.spinner(
            "01 / Investigating vulnerable source..."
        ):

            findings = scan_application(TARGET)

        stages.append(
            (
                "SOURCE ANALYSIS",
                True,
                f"{len(findings)} finding(s) detected",
            )
        )


        # ----------------------------------------------------
        # STAGE 02: PATCH GENERATION
        # ----------------------------------------------------

        with st.spinner(
            "02 / Generating remediation artefacts..."
        ):

            generate_patches()

        patch_ok = (
            PATCHED.exists()
            and DIFF.exists()
        )

        stages.append(
            (
                "PATCH GENERATION",
                patch_ok,
                "Patched source and unified diff checked",
            )
        )

        if not patch_ok:

            raise RuntimeError(
                "Patch engine did not generate expected artefacts."
            )


        # ----------------------------------------------------
        # STAGE 03: REGRESSION TESTING
        # ----------------------------------------------------

        with st.spinner(
            "03 / Executing regression test suite..."
        ):

            result = subprocess.run(
                [
                    sys.executable,
                    "-m",
                    "pytest",
                    "-v",
                ],

                cwd=ROOT,
                capture_output=True,
                text=True,
                timeout=180,
            )

        tests_ok = result.returncode == 0

        stages.append(
            (
                "REGRESSION VALIDATION",
                tests_ok,
                f"pytest exit code: {result.returncode}",
            )
        )


        # ----------------------------------------------------
        # STAGE 04: REPORT GENERATION
        # ----------------------------------------------------

        if tests_ok:

            with st.spinner(
                "04 / Recording investigation evidence..."
            ):

                generate_report()

            stages.append(
                (
                    "REPORT GENERATION",
                    REPORT.exists(),
                    "JSON assessment refreshed",
                )
            )

        else:

            stages.append(
                (
                    "REPORT GENERATION",
                    False,
                    "Skipped because regression tests failed",
                )
            )


        # ----------------------------------------------------
        # UPDATE SESSION
        # ----------------------------------------------------

        st.session_state.findings = findings
        st.session_state.test_result = result
        st.session_state.stages = stages

        st.session_state.complete = True

        st.session_state.last_run = datetime.now().strftime(
            "%d %b %Y · %H:%M:%S"
        )

        st.session_state.error = None

    except Exception as exc:

        st.session_state.stages = stages

        st.session_state.error = (
            f"{type(exc).__name__}: {exc}"
        )

        st.session_state.complete = True

    st.rerun()

from pathlib import Path


PROJECT_ROOT = Path(__file__).parent.parent

SOURCE_FILE = PROJECT_ROOT / "vulnerable_app" / "app.py"
PATCHED_FILE = PROJECT_ROOT / "vulnerable_app" / "patched_app.py"
DIFF_FILE = PROJECT_ROOT / "reports" / "security_patch.diff"


def generate_patches():

    original_code = SOURCE_FILE.read_text(encoding="utf-8")
    patched_code = original_code

    # PATCH 1: SQL Injection
    replacements = [
        (
            'query = f"SELECT * FROM users WHERE username = \'{username}\'"',
            'query = "SELECT * FROM users WHERE username = ?"'
        ),

        (
            'results = connection.execute(query).fetchall()',
            'results = connection.execute(query, (username,)).fetchall()'
        ),

        # PATCH 2: IDOR — use authenticated session identity
        (
            'from flask import Flask, jsonify, request',
            'from flask import Flask, jsonify, request, session'
        ),

        (
            'user_id = request.args.get("user_id", type=int)',
            'user_id = session.get("user_id")\n\n'
            '    if user_id is None:\n'
            '        return jsonify({"error": "Authentication required"}), 401'
        ),

        (
            '"SELECT * FROM orders WHERE id = ?",\n        (order_id,)',
            '"SELECT * FROM orders WHERE id = ? AND user_id = ?",\n'
            '        (order_id, user_id)'
        ),

        (
            'app = Flask(__name__)',
            'app = Flask(__name__)\n'
            'app.secret_key = "local-demo-only-change-this"'
        )
    ]

    # Verify that all expected vulnerable patterns exist
    for old, new in replacements:
        if old not in patched_code:
            raise ValueError(
                f"Could not find expected code to patch:\n{old}"
            )

    # Apply patches
    for old, new in replacements:
        patched_code = patched_code.replace(old, new, 1)

    # Save patched application
    PATCHED_FILE.write_text(
        patched_code,
        encoding="utf-8"
    )

    # Generate before/after diff
    diff = difflib.unified_diff(
        original_code.splitlines(),
        patched_code.splitlines(),
        fromfile="vulnerable_app/app.py",
        tofile="vulnerable_app/patched_app.py",
        lineterm=""
    )

    DIFF_FILE.write_text(
        "\n".join(diff),
        encoding="utf-8"
    )

    print("PATCHES GENERATED SUCCESSFULLY!")
    print(f"Patched application: {PATCHED_FILE}")
    print(f"Patch diff: {DIFF_FILE}")

import ast
from pathlib import Path


def detect_sql_injection(file_path):
    findings = []

    source = Path(file_path).read_text(encoding="utf-8")
    tree = ast.parse(source)

    for function in ast.walk(tree):
        if not isinstance(function, ast.FunctionDef):
            continue

        f_string_variables = {}

        for node in ast.walk(function):
            if isinstance(node, ast.Assign):
                if isinstance(node.value, ast.JoinedStr):
                    for target in node.targets:
                        if isinstance(target, ast.Name):
                            f_string_variables[target.id] = node.lineno

            if isinstance(node, ast.Call):
                if (
                    isinstance(node.func, ast.Attribute)
                    and node.func.attr == "execute"
                    and node.args
                ):
                    argument = node.args[0]

                    if isinstance(argument, ast.JoinedStr):
                        findings.append({
                            "type": "SQL Injection",
                            "severity": "High",
                            "line": node.lineno,
                            "evidence": "SQL query uses an f-string.",
                            "file": str(file_path)
                        })

                    elif (
                        isinstance(argument, ast.Name)
                        and argument.id in f_string_variables
                    ):
                        findings.append({
                            "type": "SQL Injection",
                            "severity": "High",
                            "line": f_string_variables[argument.id],
                            "evidence": "An f-string SQL query is passed to execute().",
                            "file": str(file_path)
                        })

    return findings


def detect_idor(file_path):
    findings = []

    source = Path(file_path).read_text(encoding="utf-8")
    tree = ast.parse(source)

    for function in ast.walk(tree):
        if not isinstance(function, ast.FunctionDef):
            continue

        function_source = ast.get_source_segment(source, function)

        reads_user_id = (
            "request.args.get" in function_source
            and "user_id" in function_source
        )

        if not reads_user_id:
            continue

        for node in ast.walk(function):
            if not isinstance(node, ast.Call):
                continue

            if not (
                isinstance(node.func, ast.Attribute)
                and node.func.attr == "execute"
                and node.args
                and isinstance(node.args[0], ast.Constant)
                and isinstance(node.args[0].value, str)
            ):
                continue

            sql = node.args[0].value.lower()

            if (
                "from orders" in sql
                and "where" in sql
                and "user_id" not in sql
            ):
                findings.append({
                    "type": "Broken Access Control (IDOR)",
                    "severity": "High",
                    "line": node.lineno,
                    "evidence": "Order query does not check order ownership.",
                    "file": str(file_path)
                })

    return findings

from vulnerable_app.patched_app import app, initialize_database

def client():

def authenticate(client, user_id):
    with client.session_transaction() as session:
        session["user_id"] = user_id


def test_normal_search_still_works(client):

    response = client.get("/search?username=alice")

    assert response.status_code == 200
    assert len(response.json) == 1
    assert response.json[0]["username"] == "alice"


def test_sql_injection_is_prevented(client):

    payload = "' OR '1'='1' -- "

    response = client.get(
        "/search",
        query_string={"username": payload}
    )

    assert response.status_code == 200
    assert len(response.json) == 0

    print("\nSQL INJECTION PATCH VERIFIED!")


def test_authenticated_user_can_access_own_order(client):

    response = client.get("/orders/101")

    assert response.status_code == 200

def test_user_cannot_access_another_users_order(client):

    assert response.status_code == 404

    print("\nIDOR PATCH VERIFIED!")

import sqlite3
from pathlib import Path

from flask import Flask, jsonify, request

app = Flask(__name__)

DATABASE = Path(__file__).parent / "patchproof.db"

# Create a database with synthetic users and orders
def initialize_database():
    connection = sqlite3.connect(DATABASE)

    cursor = connection.cursor()

    cursor.execute("""
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY,
            username TEXT NOT NULL,
            email TEXT NOT NULL
        )
    """)

    cursor.execute("""
        CREATE TABLE IF NOT EXISTS orders (
            id INTEGER PRIMARY KEY,
            user_id INTEGER NOT NULL,
            item TEXT NOT NULL,
            amount REAL NOT NULL
        )
    """)

    cursor.execute("DELETE FROM users")
    cursor.execute("DELETE FROM orders")

    cursor.executemany(
        "INSERT INTO users VALUES (?, ?, ?)",
        [
            (1, "alice", "alice@example.com"),
            (2, "bob", "bob@example.com"),
            (3, "charlie", "charlie@example.com")
        ]
    )

    cursor.executemany(
        "INSERT INTO orders VALUES (?, ?, ?, ?)",
        [
            (101, 1, "Laptop", 55000),
            (102, 1, "Keyboard", 1500),
            (201, 2, "Headphones", 3000),
            (301, 3, "Monitor", 12000)
        ]
    )

    connection.commit()
    connection.close()

# Connect to the SQLite database
def get_database():
    connection = sqlite3.connect(DATABASE)
    connection.row_factory = sqlite3.Row
    return connection

@app.route("/")
def home():
    return jsonify({
        "application": "PATCHPROOF Vulnerable Application",
        "status": "running",
        "endpoints": [
            "/search?username=alice",
            "/orders/101?user_id=1"
        ]
    })

# VULNERABILITY 1: SQL Injection
@app.route("/search")
def search_user():
    username = request.args.get("username", "")

    connection = get_database()

    # INTENTIONALLY VULNERABLE:
    # User input is directly inserted into SQL.
    query = f"SELECT * FROM users WHERE username = '{username}'"

    try:
        results = connection.execute(query).fetchall()

        return jsonify([dict(row) for row in results])

    except sqlite3.Error as error:
        return jsonify({"error": str(error)}), 400

    finally:
        connection.close()

# VULNERABILITY 2: Broken Access Control (IDOR)
@app.route("/orders/<int:order_id>")
def get_order(order_id):
    user_id = request.args.get("user_id", type=int)

    connection = get_database()

    # INTENTIONALLY VULNERABLE:
    # The requested order is fetched without checking ownership.
    order = connection.execute(
        "SELECT * FROM orders WHERE id = ?",
        (order_id,)
    ).fetchone()

    connection.close()

    if order is None:
        return jsonify({"error": "Order not found"}), 404

    return jsonify({
        "requested_by": user_id,
        "order": dict(order)
    })

if __name__ == "__main__":
    initialize_database()

    app.run(
        host="127.0.0.1",
        port=5000,
        debug=False
    )

# Create a database with synthetic users and orders
def initialize_database():

            id INTEGER PRIMARY KEY,

            id INTEGER PRIMARY KEY,

        )

    cursor.executemany(
        "INSERT INTO users VALUES (?, ?, ?)",
        [
            (1, "alice", "alice@example.com"),

# Connect to the SQLite database
def get_database():
    connection = sqlite3.connect(DATABASE)

def home():
